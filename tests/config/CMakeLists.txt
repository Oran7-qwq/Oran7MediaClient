cmake_minimum_required(VERSION 3.16)
project(ConfigChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Qml)
set(ROOT "${ORAN7_SOURCE_ROOT}")
set(SERVICE "${ROOT}/src/cpp/Oran7MediaClient")
add_executable(config_checks main.cpp "${SERVICE}/Oran7ConfigBinding.cpp")
target_include_directories(config_checks PRIVATE "${ROOT}/src/cpp/GlobalSingleton" "${ROOT}/src/cpp/Oran7Config" "${SERVICE}")
target_compile_definitions(config_checks PRIVATE APP_ROOT="${ROOT}")
target_link_libraries(config_checks PRIVATE Oran7Core Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Test)
