cmake_minimum_required(VERSION 3.16)
project(SettingsAnimationChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick Test)
set(ROOT "${ORAN7_SOURCE_ROOT}")
add_executable(animation_checks main.cpp icons.qrc)
target_compile_definitions(animation_checks PRIVATE APP_ROOT="${ROOT}")
target_link_libraries(animation_checks PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Test)
